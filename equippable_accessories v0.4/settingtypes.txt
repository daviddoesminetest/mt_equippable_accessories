
# [preset for the Equippable Accessory Inventory]
# 1 - Default: there's a lot of accessory slots and a different slot for every accessory type. Dyes and vanity have their own pages.
# 2 - Terraria-like: there's 5 slots that can accept any accessory. Accessories, vanity and dyes are in the same page.
# 3 - Terraria-like, with pages: almost the same as '2', but accessories, vanity and dyes have separate pages.
# 4 - Empty: like default, but comes with no accessory slots. Only use if you want to purely register your own slots.
# 5 - Custom: use it if you are going to make your own inventory by replacing 'eqp_accessories.form(player)' in 'init.lua'.
# 6 - None: there's no inventory. Use it if you want to make and handle the inventory completely by yourself.
eqp_acc_inv_preset (Inventory Preset) int 1 1 6

# If set to false, armors and armor inventory from Equippable Accessories will be disabled, only leaving the actual accessories
# (Equippable Accessories armor is forcefully disabled if using 3D Armor or if used on Mineclonia)
eqp_acc_armor (Enable Armor) bool true

# If set to false, "Health Boost" stat will not be registered
# (use it if it is incompatible with Max HP change from another mod, or if you are tired of Mineclonia occasionally whining about health disagreements.)
eqp_acc_health (Use Health Boost Stat) bool true

# [sets the durability mode for accessories]
# 0 - Disabled. Accessories never wear out
# 1 - Enabled.
# 2 - Enabled and wear caused by player getting hurt, scales with how high the damage was.
eqp_acc_durability (Durability Mode) int 2 0 2

# with defense type '2', defines the needed amount of defense to reduce damage by 1 point
# with defense type '1', reduces the protection % defense formula gives you.
eqp_acc_def_per_dmg (Defense Per Damage) int 5 1 10000

# 1: uses a formula, that results in % damage reduction, that can keep increasing, but never reach 100%: 'floor((<Def>/(<Def>+(4*<Def Per DMG>)))*10000)/10000'
# 2: for every number of defense, determined by 'Defense Per Damage' setting, player takes 1 point of damage less
eqp_acc_deftype (Defense Type) int 1 1 2

# defines the minimum damage player will still take, even if they should be fully protected by accessories
eqp_acc_mindmg (Minimum Damage) float 0.4 0 1

# Expands the space in inventory of bones, for better odds of fitting all accessories inside, if stored inside bones on death.
eqp_acc_expandbones (Expand Bones Inventory) bool true

# [determines whether or not accessories drop on death]
# === optional ===
#     depends either on MTG bones mod and drops, dosen't drop or is stored in bones based on bones settings, 
#     or on MCL settings and drops if 'Keep inventory on death' setting is not enabled.
#
# === forced ===
#     drops regardless of game or death drop settings         
#
# === disabled ===          
#     does not drop on death, regardless of game or death drop settings
eqp_acc_deathdrop (Accessory Drop On Death) enum optional optional,forced,disabled

# if players in your game/mod are supposed to start with more or less than 20 Max HP, 
# change this setting to determine how much Max HP players get with no accessories equipped
eqp_acc_initMaxHP (Initial MaxHP) int 20 10 10000

# Shows info on stat bonuses and other effects that an accessory gives you, in the description.
eqp_acc_exdesc (Extended Description) bool true

# allows some accessories to let you fly by holding down 'space'
eqp_acc_flight (Enable Flight) bool true

# defines the interval in seconds, for equipment inventory to be auto-saved.
# (Unequipping/Equipping or breaking an accessory causes a instant save)
eqp_acc_saveInt (Save Interval) int 20 1 60

# enables accessory entity "on_step" function and registers dyes that will constantly change their look.
eqp_acc_dynamic_dyes (Dynamic dyes) bool false

# hides accessory double-dyes from creative menu, so you don't have to experience a 'moreblocks moment'.
eqp_acc_creative_double_dyes (Double-dyes in creative menu) bool true
